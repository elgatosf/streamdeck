### RegistrationInfo{#registrationinfo}

Information about the Stream Deck application, the plugin, the user's operating system, user's Stream Deck devices, etc.

```ts
type RegistrationInfo = {
	application: {
		font: string;
		language: "de" | "en" | "es" | "fr" | "ja" | "ko" | "zh_CN" | "zh_TW";
		platform: "mac" | "windows";
		platformVersion: string;
		version: string;
	};
	colors: {
		buttonMouseOverBackgroundColor: string;
		buttonPressedBackgroundColor: string;
		buttonPressedBorderColor: string;
		buttonPressedTextColor: string;
		highlightColor: string;
	};
	devicePixelRatio: number;
	devices: {
		id: string;
		name: string;
		size: {
			columns: number;
			rows: number;
		};
		type: DeviceType;
	}[];
	plugin: {
		uuid: string;
		version: string;
	};
};
```

<ObjectMember name="application" >
	<ObjectMember.Type>object</ObjectMember.Type>
	
Stream Deck application specific information.

<ObjectMember name="font">
	<ObjectMember.Type>string</ObjectMember.Type>
	Font being used by the Stream Deck application.
</ObjectMember>

<ObjectMember name="language">
	<ObjectMember.Type>"de" | "en" | "es" | "fr" | "ja" | "ko" | "zh_CN" | "zh_TW"</ObjectMember.Type>
	Users preferred language; this is used by the Stream Deck application for localization.
</ObjectMember>

<ObjectMember name="platform">
	<ObjectMember.Type>"mac" | "windows"</ObjectMember.Type>
	Operating system.
</ObjectMember>

<ObjectMember name="platformVersion">
	<ObjectMember.Type>string</ObjectMember.Type>
	Operating system version, e.g. "10" for Windows 10.
</ObjectMember>

<ObjectMember name="version">
	<ObjectMember.Type>string</ObjectMember.Type>
	Stream Deck application version.
</ObjectMember>

</ObjectMember>

<ObjectMember name="colors" >
	<ObjectMember.Type>object</ObjectMember.Type>
	
Collection of preferred colors used by the Stream Deck.

<ObjectMember name="buttonMouseOverBackgroundColor">
	<ObjectMember.Type>string</ObjectMember.Type>
	Color that denotes the background of a button that is being moused over.
</ObjectMember>

<ObjectMember name="buttonPressedBackgroundColor">
	<ObjectMember.Type>string</ObjectMember.Type>
	Color that denotes the background of a pressed button.
</ObjectMember>

<ObjectMember name="buttonPressedBorderColor">
	<ObjectMember.Type>string</ObjectMember.Type>
	Color that denotes the border of a press button.
</ObjectMember>

<ObjectMember name="buttonPressedTextColor">
	<ObjectMember.Type>string</ObjectMember.Type>
	Color that denotes the text of a pressed button.
</ObjectMember>

<ObjectMember name="highlightColor">
	<ObjectMember.Type>string</ObjectMember.Type>
	Color of highlighted text.
</ObjectMember>

</ObjectMember>

<ObjectMember name="devicePixelRatio">
	<ObjectMember.Type>number</ObjectMember.Type>
	Pixel ratio, used to identify if the Stream Deck application is running on a high DPI screen.
</ObjectMember>

<ObjectMember name="devices">
	<ObjectMember.Type>object[]</ObjectMember.Type>
	Devices associated with the Stream Deck application; this may include devices that are not currently connected. Use
	`"deviceDidConnect"` event to determine which devices are active.
</ObjectMember>

<ObjectMember name="plugin" >
	<ObjectMember.Type>object</ObjectMember.Type>
	
Information about the plugin.

<ObjectMember name="uuid">
	<ObjectMember.Type>string</ObjectMember.Type>
	Unique identifier of the plugin, as defined by the plugin.
</ObjectMember>

<ObjectMember name="version">
	<ObjectMember.Type>string</ObjectMember.Type>
	Version of the plugin.
</ObjectMember>

</ObjectMember>
